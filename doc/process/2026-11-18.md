- **Giải pháp**:

    - Các yêu cầu về cấu trúc JSON, lưu ý an toàn và phiếu `usage_profile` được chuyển xuống phần chỉ thị kỹ thuật phía dưới.
  - Kết quả: Tỷ lệ Gemini kích hoạt công cụ tìm kiếm Google tăng rõ rệt, bóc tách được các số liệu từ các nguồn chính thống (Đại học Nông Lâm, Viện BVTV, Khuyến nông).

  - Trong CSDL `Products`, hệ thống lưu trữ kết hợp mô tả đặc tính và liều lượng sử dụng trong trường `DosageInstruction` nhằm bảo đảm tính tương thích với Trợ lý AI Lavi.
  - Trước đây, thuật toán tách dữ liệu dựa vào dấu xuống dòng cuối cùng (`lastIndex`). Do đó, khi hướng dẫn sử dụng gồm nhiều dòng (ví dụ: dòng 1: Liều lượng, dòng 2: Cách phun, dòng 3: Thời điểm), các dòng trên bị cắt nhầm sang trường Mô tả.

    export function combineDescriptionDosage(description: string, dosage: string): string {
      const cleanDesc = description.trim();
      const cleanDosage = dosage.trim();
      if (!cleanDesc) return cleanDosage;
      if (!cleanDosage) return cleanDesc;
      return `${cleanDesc}\n\nHướng dẫn sử dụng:\n${cleanDosage}`;
    }

    export function splitDescriptionDosage(fullText: string): { description: string; dosage: string } {
      const marker = 'Hướng dẫn sử dụng:';
      const markerIdx = fullText.indexOf(marker);
      if (markerIdx !== -1) {
        return {
          description: fullText.substring(0, markerIdx).trim(),
          dosage: fullText.substring(markerIdx + marker.length).trim(),
        };
